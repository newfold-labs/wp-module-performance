5/**
6 * Authoritative, cached check for whether the Redis service (daemon) is available on this
7 * site's server, per the hosting API (HUAPI GET /performance/redis -> HAL `daemon_active`).
8 *
9 * This exists because the phpredis PHP extension being loaded is NOT a reliable signal: on some
10 * server generations (e.g. legacy CentOS 7 / hostmonster boxes) the extension is present but the
11 * Redis daemon was never deployed, so offering the object-cache UI there produces "Could not
12 * enable object cache" errors. Only the server-side daemon status can tell those boxes apart.
13 *
14 * The result is cached in a transient so the settings page / runtime SDK does not make a network
15 * call on every render.
16 */
17final class RedisServiceAvailability {
18

22     * @var string

24    const TRANSIENT_KEY = 'nfd_performance_redis_service_available';

28     * Redis today is very unlikely to lose it.
29     *
30     * @var int
31     */
32    const TTL_AVAILABLE = 43200; // 12 hours.

37     *

40    const TTL_UNAVAILABLE = 3600; // 1 hour.
