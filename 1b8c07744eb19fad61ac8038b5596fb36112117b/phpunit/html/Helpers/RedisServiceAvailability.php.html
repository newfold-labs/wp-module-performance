439    /**
440     * Ask the hosting API whether the Redis daemon is active on this site's server.
441     *
442     * Both calls use the short probe timeout: this runs while an admin page renders, and a probe that
443     * times out is indeterminate and retried later rather than fatal.
444     *
445     * @return bool|null True/false when the server answered definitively; null when the answer could
446     *                   not be determined (Hiive not connected, token/site missing, or a transient
447     *                   HTTP error) and the caller should not cache the result for long.
448     */

450        $timeout = SiteApisConfig::redis_probe_timeout_seconds();
451
452        $context = RedisCredentialsProvisioner::get_hosting_context( $timeout );
453        if ( is_wp_error( $context ) ) {
454            return null;
455        }
456
457        $status = HostingUapiClient::get_site_performance_redis( $context['token'], $context['site_id'], $timeout );
458
459        if ( is_wp_error( $status ) ) {
460            $data           = $status->get_error_data();
461            $customer_error = ( is_array( $data ) && isset( $data['customer_error'] ) ) ? (string) $data['customer_error'] : '';
462
463            // These customer errors mean the box definitively cannot run object cache.
464            if (
465                self::CUSTOMER_ERROR_SERVICE_INACTIVE === $customer_error
466                || self::CUSTOMER_ERROR_PHP_UNSUPPORTED === $customer_error
467            ) {
468                return false;
469            }
470
471            // Any other error (network, auth, unknown) is indeterminate.
472            return null;
473        }
474
475        return ! empty( $status['redis_service_active'] );
476    }
